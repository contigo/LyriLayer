cmake_minimum_required(VERSION 3.22.1)
project(ailyrickit_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Optimize inference even in development APKs; keep portable ARMv8 baseline.
add_compile_options($<$<COMPILE_LANGUAGE:C>:-O3> $<$<COMPILE_LANGUAGE:CXX>:-O3>)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
include(FetchContent)
FetchContent_Declare(whisper_src GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git GIT_TAG v1.7.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(whisper_src)
add_library(ailyrickit_whisper SHARED whisper_jni.cpp)
target_link_libraries(ailyrickit_whisper PRIVATE whisper android log)

# Prevent interposition with the app legacy whisper.cpp static archives.
target_link_options(ailyrickit_whisper PRIVATE "-Wl,--exclude-libs,ALL")
